// --- Gemini API (Unchanged) ---
const generateGeminiContent = async (prompt: string): Promise<string> => {
  const apiKey = ""; 
  const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;
  try {
    const response = await fetch(url, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }] })
    });
    if (!response.ok) throw new Error(`Gemini API Error`);
    const data = await response.json();
    return data.candidates?.[0]?.content?.parts?.[0]?.text || "AI unavailable.";
  } catch (error) {
    return "Connecting to AI...";
  }
};

// --- Main App ---
export default function App() {
  const [currentView, setCurrentView] = useState<View>('LEADERBOARD'); // Updated default view for demo
  const [selectedSportId, setSelectedSportId] = useState<string>('badminton');
  const [userRole, setUserRole] = useState<UserRole>('GUEST');
  const [currentUserId, setCurrentUserId] = useState<string | null>(null);
  const [matches, setMatches] = useState<Match[]>(INITIAL_MATCHES);
  const [registeredPlayers, setRegisteredPlayers] = useState<Player[]>([]);
  const [notification, setNotification] = useState<string | null>(null);
  const [systemLogs, setSystemLogs] = useState<LogEntry[]>([]);

  const activeSport = SPORTS_DATA.find(s => s.id === selectedSportId) || SPORTS_DATA[0];

  const addLog = (action: string, details: string, type: LogEntry['type'] = 'INFO') => {
    const newLog: LogEntry = {
      id: Math.random().toString(36).substr(2, 9),
      timestamp: new Date(),
      action,
      details,
      type
    };
    setSystemLogs(prev => [newLog, ...prev]);
  };

  const showNotification = (msg: string) => {
    setNotification(msg);
    setTimeout(() => setNotification(null), 4000);
  };

  const navigate = (view: View) => {
    window.scrollTo(0, 0);
    setCurrentView(view);
  };

  // Mock live updates
  useEffect(() => {
    const interval = setInterval(() => {
      setMatches(prev => prev.map(m => {
        if (m.status === 'ONGOING' && Math.random() > 0.98) {
          const winnerId = Math.random() > 0.5 ? m.playerA.id : m.playerB.id;
          const score = '21-19, 21-18'; 
          addLog('MATCH_COMPLETE', `Match ${m.id} completed. Winner: ${winnerId === m.playerA.id ? m.playerA.name : m.playerB.name}`, 'SUCCESS');
          return { ...m, status: 'COMPLETED', winnerId, score, endTime: new Date().toISOString() };
        }
        return m;
      }));
    }, 3000);
    return () => clearInterval(interval);
  }, []);

  const handleRegister = (player: Partial<Player>, sportId: string) => {
    const uniqueId = `XTH-${sportId.substring(0,2).toUpperCase()}-${Math.floor(1000 + Math.random() * 9000)}`;
    const newPlayer = { ...player, id: uniqueId, sportId, club: player.club || 'Independent' } as Player;
    setRegisteredPlayers([...registeredPlayers, newPlayer]);
    showNotification(`Registered! ID: ${uniqueId}`);
    addLog('PLAYER_REGISTER', `New player ${newPlayer.name} registered for ${sportId}`, 'INFO');
    return uniqueId;
  };

  const login = (role: UserRole, playerId?: string) => {
    setUserRole(role);
    if (role === 'PLAYER' && playerId) {
      setCurrentUserId(playerId);
      showNotification(`Welcome back!`);
      addLog('LOGIN', `Player ${playerId} logged in`, 'INFO');
      navigate('PLAYER_DASHBOARD');
    } else {
      setCurrentUserId(null);
      showNotification(`Logged in as ${role}`);
      addLog('LOGIN', `User logged in as ${role}`, 'INFO');
      navigate(role === 'ADMIN' ? 'ADMIN' : role === 'UMPIRE' ? 'UMPIRE' : 'LANDING');
    }
  };

  return (
    <div className="min-h-screen bg-gradient-to-br from-slate-50 to-slate-100 font-sans text-slate-900 selection:bg-emerald-100/50">
      {notification && (
        <div className="fixed top-24 right-6 z-50 animate-in slide-in-from-right-10 fade-in duration-300">
          <div className="bg-white/80 backdrop-blur-xl border border-white/40 shadow-2xl shadow-emerald-900/10 px-6 py-4 rounded-2xl flex items-center gap-4">
            <div className="bg-emerald-500/10 p-2 rounded-full">
              <CheckCircle className="w-5 h-5 text-emerald-600" />
            </div>
            <div>
              <p className="text-sm font-bold text-slate-800">Notification</p>
              <p className="text-xs text-slate-500 font-medium">{notification}</p>
            </div>
          </div>
        </div>
      )}

      <Navbar currentView={currentView} navigate={navigate} userRole={userRole} onLogout={() => { setUserRole('GUEST'); setCurrentUserId(null); navigate('LANDING'); }} />

      <main className="pt-24 pb-12 px-4 md:px-8 min-h-screen flex flex-col max-w-[1600px] mx-auto w-full">
        <div className="animate-in fade-in slide-in-from-bottom-4 duration-700">
          {currentView === 'LANDING' && <LandingPage navigate={navigate} />}
          {currentView === 'SPORTS_DIRECTORY' && <SportsDirectory navigate={navigate} onSelectSport={(id) => { setSelectedSportId(id); navigate('SPORT_DETAIL'); }} />}
          {currentView === 'SPORT_DETAIL' && <SportDetailPage sport={activeSport} onRegister={handleRegister} navigate={navigate} />}
          {currentView === 'FIXTURES' && <FixturesView matches={matches.filter(m => m.sportId === selectedSportId)} activeSport={activeSport} sports={SPORTS_DATA} onSportChange={setSelectedSportId} currentUserId={currentUserId} />}
          {currentView === 'SCHEDULE' && <ScheduleView matches={matches.filter(m => m.sportId === selectedSportId)} courts={MOCK_COURTS.filter(c => c.sportId === selectedSportId)} currentUserId={currentUserId} activeSport={activeSport} sports={SPORTS_DATA} onSportChange={setSelectedSportId} />}
          {currentView === 'LEADERBOARD' && <LeaderboardView activeSport={activeSport} sports={SPORTS_DATA} onSportChange={setSelectedSportId} />}
          {currentView === 'PLAYER_DASHBOARD' && <PlayerDashboard matches={matches} currentUserId={currentUserId} navigate={navigate} activeSport={activeSport} />}
          {currentView === 'ADMIN' && <AdminDashboard matches={matches} setMatches={setMatches} activeSport={activeSport} sports={SPORTS_DATA} onSportChange={setSelectedSportId} registeredPlayers={registeredPlayers} showNotification={showNotification} systemLogs={systemLogs} />}
          {currentView === 'LOGIN' && <LoginPage onLogin={login} />}
          {currentView === 'UMPIRE' && <UmpireConsole matches={matches} setMatches={setMatches} showNotification={showNotification} addLog={addLog} />}
        </div>
      </main>
    </div>
  );
}

// --- NAVBAR ---
function Navbar({ currentView, navigate, userRole, onLogout }: any) {
  return (
    <nav className="fixed top-4 left-4 right-4 h-16 bg-white/70 backdrop-blur-xl border border-white/40 rounded-2xl shadow-sm z-50 px-6 flex items-center justify-between transition-all duration-300">
      <div className="flex items-center gap-3 cursor-pointer group" onClick={() => navigate('LANDING')}>
        <div className="w-9 h-9 bg-gradient-to-tr from-emerald-500 to-emerald-400 rounded-xl flex items-center justify-center text-white font-bold text-xl shadow-lg shadow-emerald-500/20 group-hover:scale-105 transition-transform">
          X
        </div>
        <span className="text-xl font-extrabold text-slate-800 tracking-tight group-hover:text-emerald-600 transition-colors">
          Xthlete
        </span>
      </div>

      <div className="hidden md:flex items-center gap-1 bg-slate-100/50 p-1 rounded-xl border border-white/50">
        {['LANDING', 'FIXTURES', 'SCHEDULE', 'LEADERBOARD'].map((view) => (
          <button 
            key={view}
            onClick={() => navigate(view)}
            className={`px-5 py-2 rounded-lg text-sm font-bold transition-all duration-300 ${
              currentView === view 
                ? 'bg-white text-slate-800 shadow-sm scale-100' 
                : 'text-slate-500 hover:text-slate-700 hover:bg-white/50'
            }`}
          >
            {view === 'LANDING' ? 'Home' : view.charAt(0) + view.slice(1).toLowerCase()}
          </button>
        ))}
      </div>

      <div className="flex items-center gap-4">
        {userRole === 'GUEST' ? (
          <button 
            onClick={() => navigate('LOGIN')}
            className="bg-slate-900 hover:bg-slate-800 text-white px-6 py-2.5 rounded-xl text-sm font-bold transition-all shadow-lg shadow-slate-900/20 hover:-translate-y-0.5"
          >
            Sign In
          </button>
        ) : (
          <div className="flex items-center gap-3 pl-4 border-l border-slate-200">
             <div className="text-right hidden sm:block">
                <p className="text-xs font-bold text-slate-800">{userRole}</p>
                <p className="text-[10px] font-medium text-emerald-600">Online</p>
             </div>
             <div className="w-9 h-9 rounded-full bg-gradient-to-br from-emerald-400 to-cyan-500 text-white flex items-center justify-center font-bold text-sm shadow-md ring-2 ring-white">
                {userRole[0]}
             </div>
             <button onClick={onLogout} className="p-2 hover:bg-red-50 text-slate-400 hover:text-red-500 rounded-lg transition-colors">
                <LogOut className="w-4 h-4" />
             </button>
          </div>
        )}
      </div>
    </nav>
  );
}

function FixturesView({ matches, activeSport, sports, onSportChange, currentUserId }: any) {
  const [activeFormat, setActiveFormat] = useState<Format>('knockout');

  const KnockoutBracket = () => {
    const rounds = useMemo(() => {
      const r: Record<number, Match[]> = {};
      matches.forEach((m: Match) => {
        if (!r[m.round]) r[m.round] = [];
        r[m.round].push(m);
      });
      Object.keys(r).forEach(k => r[parseInt(k)].sort((a, b) => (a.matchNumber || 0) - (b.matchNumber || 0)));
      return r;
    }, [matches]);

    return (
      <div className="overflow-x-auto pb-12 pt-4 relative min-h-[600px]">
        {/* Connector Lines (Styled) */}
        <svg className="absolute inset-0 pointer-events-none w-full h-full z-0 text-slate-300/60" style={{ minWidth: '900px' }}>

          <path d="M 340 670 H 360 V 580 H 380" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" />

          <path d="M 680 580 H 700 V 400 H 720" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" />
        </svg>

            return (
              <div key={roundNum} className={`flex flex-col ${spacingClass} min-w-[340px]`}>
                <div className="text-center">

function MatchCard({ match, currentUserId, sportName }: { match: Match, currentUserId: string | null, sportName: string }) {
  const isUserMatch = currentUserId && (match.playerA.id === currentUserId || match.playerB.id === currentUserId);
  const [analysis, setAnalysis] = useState<string | null>(null);
  const [loading, setLoading] = useState(false);
  const [showAnalysis, setShowAnalysis] = useState(false);

  const handleAnalyze = async (e: React.MouseEvent) => {
    e.stopPropagation();
    if (analysis) { setShowAnalysis(true); return; }
    setLoading(true); setShowAnalysis(true);
    const prompt = `Act as an excited sports commentator. Write a short, 2-sentence hype prediction for ${match.playerA.name} vs ${match.playerB.name}.`;
    const result = await generateGeminiContent(prompt);
    setAnalysis(result);
    setLoading(false);
  };
  
  return (
    <div className={`relative group w-full transition-transform duration-300 hover:-translate-y-1 ${isUserMatch ? 'ring-2 ring-emerald-400 rounded-2xl shadow-emerald-100' : ''}`}>
      <div className="bg-white/80 backdrop-blur-xl border border-white/60 rounded-2xl p-0 shadow-sm hover:shadow-xl hover:shadow-slate-200/60 transition-all relative overflow-hidden">
        
        <button onClick={handleAnalyze} className="absolute top-3 right-3 w-8 h-8 flex items-center justify-center rounded-full bg-indigo-50 text-indigo-500 hover:bg-indigo-500 hover:text-white transition-all z-20 opacity-0 group-hover:opacity-100 shadow-sm">
          <Sparkles className="w-4 h-4" />
        </button>

        {/* Player A */}
        <div className={`px-6 py-4 flex justify-between items-center border-b border-slate-50 ${match.winnerId === match.playerA.id ? 'bg-gradient-to-r from-emerald-50/50 to-transparent' : ''}`}>
          <div>
            <div className={`font-bold text-lg leading-tight ${match.winnerId === match.playerA.id ? 'text-slate-900' : 'text-slate-600'}`}>
              {match.playerA.name}
            </div>
            <div className="text-[11px] font-bold text-slate-400 uppercase tracking-wider mt-1">{match.playerA.club}</div>
          </div>
          {match.status === 'COMPLETED' && match.winnerId === match.playerA.id && (
            <div className="bg-emerald-500 text-white p-1 rounded-full shadow-sm shadow-emerald-200"><Check className="w-3 h-3" /></div>
          )}
        </div>

        {/* Player B */}
        <div className={`px-6 py-4 flex justify-between items-center ${match.winnerId === match.playerB.id ? 'bg-gradient-to-r from-emerald-50/50 to-transparent' : ''}`}>
          <div>
            <div className={`font-bold text-lg leading-tight ${match.winnerId === match.playerB.id ? 'text-slate-900' : 'text-slate-600'}`}>
              {match.playerB.name}
            </div>
            <div className="text-[11px] font-bold text-slate-400 uppercase tracking-wider mt-1">{match.playerB.club}</div>
          </div>
          {match.status === 'COMPLETED' && match.winnerId === match.playerB.id && (
            <div className="bg-emerald-500 text-white p-1 rounded-full shadow-sm shadow-emerald-200"><Check className="w-3 h-3" /></div>
          )}
        </div>

        {/* Footer */}
        <div className="px-6 py-3 bg-slate-50/80 border-t border-slate-100 flex justify-between items-center">
          <span className={`px-2.5 py-1 rounded-md text-[10px] font-black uppercase tracking-wider ${
            match.status === 'COMPLETED' ? 'bg-slate-800 text-white' : 
            match.status === 'ONGOING' ? 'bg-emerald-100 text-emerald-700 animate-pulse' : 
            'bg-slate-200 text-slate-500'
          }`}>
            {match.status}
          </span>
          
          {match.score ? (
            <span className="font-mono font-bold text-slate-700 text-sm">{match.score}</span>
          ) : (
            <span className="text-xs font-bold text-slate-400 flex items-center gap-1.5">
              <Clock className="w-3.5 h-3.5" /> {match.startTime ? new Date(match.startTime).toLocaleTimeString([], {hour:'2-digit', minute:'2-digit'}) : 'TBA'}
            </span>
          )}
        </div>

        {/* AI Overlay */}
        {showAnalysis && (
          <div className="absolute inset-0 bg-white/90 backdrop-blur-md z-30 p-6 flex flex-col items-center justify-center text-center animate-in fade-in zoom-in-95 duration-300">
            <div className="flex items-center gap-2 text-indigo-600 font-bold text-xs uppercase tracking-wider mb-3">
              <Sparkles className="w-4 h-4" /> Gemini Insight
            </div>
            {loading ? (
              <Loader2 className="w-8 h-8 text-slate-300 animate-spin" />
            ) : (
              <p className="text-sm font-medium text-slate-600 leading-relaxed italic max-w-[240px]">"{analysis}"</p>
            )}
            <button onClick={(e) => { e.stopPropagation(); setShowAnalysis(false); }} className="mt-6 text-xs font-bold text-slate-400 hover:text-slate-600 transition-colors">DISMISS</button>
          </div>
        )}
      </div>
    </div>
  );
}

function UmpireConsole({ matches, setMatches, showNotification, addLog }: any) {
  const [selectedMatch, setSelectedMatch] = useState<Match | null>(null);
  const activeMatches = matches.filter((m: Match) => m.status === 'SCHEDULED' || m.status === 'ONGOING');

  const startMatch = (id: string) => {
    setMatches((prev: any[]) => prev.map(m => m.id === id ? { ...m, status: 'ONGOING' } : m));
    showNotification("Match Started");
    addLog('MATCH_START', `Match ${id} started by umpire`, 'INFO');
    setSelectedMatch(null);
  };

  const endMatch = (id: string, winnerId: string) => {
    setMatches((prev: any[]) => prev.map(m => m.id === id ? { ...m, status: 'COMPLETED', winnerId, score: '21-15, 21-18' } : m));
    showNotification("Score Submitted successfully");
    addLog('MATCH_END', `Match ${id} ended manually. Winner: ${winnerId}`, 'SUCCESS');
    setSelectedMatch(null);
  };

  return (
    <div className="container mx-auto px-4 py-12">
      <h2 className="text-4xl font-black text-slate-900 mb-8">Umpire Console</h2>
      
      {!selectedMatch ? (
        <div className="grid gap-6 md:grid-cols-2 lg:grid-cols-3">
          {activeMatches.map((match: Match) => (
            <div key={match.id} className="bg-white p-8 rounded-3xl border border-slate-100 shadow-xl shadow-slate-200/50 hover:shadow-2xl hover:-translate-y-1 transition-all cursor-pointer group" onClick={() => setSelectedMatch(match)}>
              <div className="flex justify-between items-center mb-6">
                <span className={`px-3 py-1 rounded-lg text-xs font-bold uppercase tracking-wider ${match.status === 'ONGOING' ? 'bg-emerald-100 text-emerald-700 animate-pulse' : 'bg-slate-100 text-slate-600'}`}>
                  {match.status}
                </span>
                <span className="text-xs text-slate-400 font-mono bg-slate-50 px-2 py-1 rounded">#{match.id.toUpperCase()}</span>
              </div>
              
              <div className="space-y-4 mb-6">
                <div className="flex justify-between items-center">
                  <span className="font-bold text-lg text-slate-800">{match.playerA.name}</span>
                </div>
                <div className="text-center text-xs font-bold text-slate-300 uppercase tracking-[0.2em]">VS</div>
                <div className="flex justify-between items-center">
                  <span className="font-bold text-lg text-slate-800">{match.playerB.name}</span>
                </div>
              </div>
              
              <div className="flex items-center gap-2 text-sm font-bold text-slate-500 bg-slate-50 p-3 rounded-xl">
                 <MapPin className="w-4 h-4 text-emerald-500" /> 
                 {MOCK_COURTS.find(c => c.id === match.courtId)?.name || 'Unassigned'}
              </div>
            </div>
          ))}
          {activeMatches.length === 0 && (
            <div className="col-span-full bg-white rounded-3xl p-12 text-center border border-dashed border-slate-300">
              <p className="text-slate-500 font-medium text-lg">No active matches assigned to you.</p>
            </div>
          )}
        </div>
      ) : (
        <div className="max-w-2xl mx-auto bg-white rounded-3xl shadow-2xl border border-slate-100 overflow-hidden">
          <div className="bg-slate-900 text-white p-8 flex justify-between items-center">
            <h3 className="font-bold text-2xl">Match Control</h3>
            <button onClick={() => setSelectedMatch(null)} className="text-slate-400 hover:text-white font-bold text-sm bg-white/10 px-4 py-2 rounded-lg hover:bg-white/20 transition-colors">Close</button>
          </div>
          <div className="p-10">
            <div className="flex justify-between items-center mb-10">
               <div className="text-center w-1/3">
                 <div className="text-2xl font-black text-slate-900 leading-tight">{selectedMatch.playerA.name}</div>
                 <div className="text-sm font-bold text-slate-400 mt-2 uppercase">{selectedMatch.playerA.club}</div>
               </div>
               <div className="text-4xl font-black text-slate-200">VS</div>
               <div className="text-center w-1/3">
                 <div className="text-2xl font-black text-slate-900 leading-tight">{selectedMatch.playerB.name}</div>
                 <div className="text-sm font-bold text-slate-400 mt-2 uppercase">{selectedMatch.playerB.club}</div>
               </div>
            </div>

            {selectedMatch.status === 'SCHEDULED' && (
              <button 
                onClick={() => startMatch(selectedMatch.id)}
                className="w-full py-5 bg-emerald-500 hover:bg-emerald-600 text-white rounded-2xl font-bold text-xl flex items-center justify-center gap-3 shadow-lg shadow-emerald-500/30 transition-transform active:scale-[0.98]"
              >
                <Play className="w-6 h-6 fill-current" /> Start Match
              </button>
            )}

            {selectedMatch.status === 'ONGOING' && (
              <div className="space-y-6">
                <div className="bg-slate-50 p-6 rounded-2xl text-center border border-slate-100">
                  <div className="text-xs font-bold text-slate-400 uppercase tracking-widest mb-2">Live Score</div>
                  <div className="text-5xl font-mono font-black text-slate-900 tracking-tight">18 - 18</div>
                </div>
                <div className="grid grid-cols-2 gap-4">
                  <button 
                    onClick={() => endMatch(selectedMatch.id, selectedMatch.playerA.id)}
                    className="py-4 bg-slate-900 text-white rounded-xl font-bold hover:bg-slate-800 shadow-lg hover:-translate-y-1 transition-all"
                  >
                    {selectedMatch.playerA.name} Wins
                  </button>
                  <button 
                    onClick={() => endMatch(selectedMatch.id, selectedMatch.playerB.id)}
                    className="py-4 bg-slate-900 text-white rounded-xl font-bold hover:bg-slate-800 shadow-lg hover:-translate-y-1 transition-all"
                  >
                    {selectedMatch.playerB.name} Wins
                  </button>
                </div>
              </div>
            )}
          </div>
        </div>
      )}
    </div>
  );
}
