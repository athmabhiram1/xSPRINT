const Button = ({ children, variant = 'primary', className = '', onClick, icon: Icon }) => {
  const baseStyle = "inline-flex items-center justify-center px-6 py-3 rounded-full font-semibold transition-all duration-200 active:scale-95 text-sm sm:text-base";
  const variants = {
    primary: `bg-[#00E18B] text-[#0A0A0A] hover:bg-[#00c47a] shadow-lg shadow-[#00E18B]/20`,
    secondary: "bg-[#0A0A0A] text-white hover:bg-gray-800 shadow-lg",
    outline: "border-2 border-[#0A0A0A] text-[#0A0A0A] hover:bg-gray-50",
    ghost: "text-gray-600 hover:bg-gray-100",
    danger: "bg-red-500 text-white hover:bg-red-600"
  };

  return (
    <button onClick={onClick} className={`${baseStyle} ${variants[variant]} ${className}`}>
      {Icon && <Icon size={18} className="mr-2" />}
      {children}
    </button>
  );
};

const Badge = ({ children, variant = 'default' }) => {
  const styles = {
    default: "bg-gray-100 text-gray-800",
    success: "bg-green-100 text-green-800",
    live: "bg-red-100 text-red-600 animate-pulse",
    outline: "border border-gray-200 text-gray-600"
  };
  return (
    <span className={`px-2.5 py-0.5 rounded-md text-xs font-bold uppercase tracking-wide ${styles[variant]}`}>
      {children}
    </span>
  );
};

const UmpirePanelView = () => {
  const [unlocked, setUnlocked] = useState(false);
  const [scores, setScores] = useState({ p1: 0, p2: 0 });
  const [set, setSet] = useState(1);
  const [code, setCode] = useState("");

  if (!unlocked) {
    return (
      <div className="min-h-[60vh] flex items-center justify-center px-4">
        <div className="max-w-md w-full bg-white p-8 rounded-3xl shadow-2xl border border-gray-100 text-center">
          <div className="w-16 h-16 bg-gray-100 rounded-2xl flex items-center justify-center mx-auto mb-6">
            <Shield className="text-[#0A0A0A]" size={32} />
          </div>
          <h2 className="text-2xl font-bold text-[#0A0A0A] mb-2">Umpire Access</h2>
          <p className="text-gray-500 mb-8">Enter the 6-digit match code to unlock the scoring panel.</p>
          
          <input 
             type="text" 
             value={code}
             onChange={(e) => setCode(e.target.value)}
             className="w-full text-center text-3xl font-mono tracking-widest py-4 border-b-2 border-gray-200 focus:border-[#00E18B] outline-none bg-transparent mb-8 text-[#0A0A0A]"
             placeholder="000000"
             maxLength={6}
          />
          
          <Button className="w-full py-4" onClick={() => code.length === 6 && setUnlocked(true)}>Unlock Match</Button>
        </div>
      </div>
    );
  }

  return (
    <div className="max-w-lg mx-auto px-4 py-8">
      <div className="bg-[#0A0A0A] text-white p-6 rounded-t-3xl flex justify-between items-center">
        <div>
           <div className="text-xs text-gray-400 uppercase tracking-widest mb-1">Court 1 • Match 101</div>
           <div className="font-bold text-xl">Mixed Doubles Final</div>
        </div>
        <div className="text-right">
           <div className="text-2xl font-bold text-[#00E18B]">Set {set}</div>
        </div>
      </div>
      
      <div className="bg-white shadow-xl rounded-b-3xl overflow-hidden border border-t-0 border-gray-200">
        {['Team Alpha', 'Team Beta'].map((team, idx) => (
           <div key={team} className={`p-8 ${idx === 0 ? 'border-b border-gray-100' : ''}`}>
              <div className="flex justify-between items-center mb-6">
                 <h3 className="text-2xl font-bold text-gray-900">{team}</h3>
                 <div className="text-6xl font-bold font-mono text-[#0A0A0A]">
                   {idx === 0 ? scores.p1 : scores.p2}
                 </div>
              </div>
              <div className="flex gap-4">
                 <button 
                   onClick={() => idx === 0 ? setScores(s => ({...s, p1: Math.max(0, s.p1-1)})) : setScores(s => ({...s, p2: Math.max(0, s.p2-1)}))}
                   className="flex-1 py-4 bg-gray-100 rounded-xl text-gray-600 font-bold hover:bg-gray-200 active:scale-95 transition-transform"
                 >
                   -1
                 </button>
                 <button 
                   onClick={() => idx === 0 ? setScores(s => ({...s, p1: s.p1+1})) : setScores(s => ({...s, p2: s.p2+1}))}
                   className="flex-[2] py-4 bg-[#00E18B] rounded-xl text-[#0A0A0A] font-bold text-xl hover:brightness-95 active:scale-95 transition-transform shadow-lg shadow-emerald-200"
                 >
                   +1 Point
                 </button>
              </div>
           </div>
        ))}
        
        <div className="p-6 bg-gray-50 flex gap-4">
           <Button variant="outline" className="flex-1 border-gray-300 bg-white" onClick={() => { setScores({p1:0, p2:0}); setSet(s => s+1); }}>Next Set</Button>
           <Button className="flex-1 bg-[#0A0A0A] text-white">End Match</Button>
        </div>
      </div>
    </div>
  );
};
